Listings

app/api/hackathons/route.tstypescript
// GET /api/hackathons
// Filters: q, city, mode, type, theme, status, fee, team, eligibility,
//          beginner, prize, within, km, sort, lat, lng, page, limit
// Auth: none  Rate: 120/min
export const GET = route(async ({ req }) => {
  const { lat, lng, page, limit } = readQuery(req, Query.passthrough());
  const filters = parseFilters(params);
  const origin = lat && lng ? { lat, lng } : null;
  const { items, nowIso } = await getHackathons();
  const results = applyFilters(items, filters, origin);
  return { nowIso, total, page, pages, filters, items: results.slice(start, end) };
}, { auth: 'none', rate: { limit: 120, windowMs: 60_000 } });
app/api/hackathons/[slug]/route.tstypescript
// GET /api/hackathons/:slug
// Returns: { nowIso, item: HackathonView, similar: HackathonView[] }
// Auth: none  Rate: 120/min
export const GET = route(async ({ params }) => {
  const found = await getHackathon(params.slug);
  if (!found) throw notFound('That hackathon');
  return { nowIso: found.nowIso, item: found.item, similar: similarTo(found.item, found.items) };
}, { auth: 'none', rate: { limit: 120, windowMs: 60_000 } });

Smart Search

app/api/search/route.tstypescript
// POST /api/search  Body: { q: string }
// "free AI hackathon in Pune this month for beginners" → Explore filters
// Uses Claude when ANTHROPIC_API_KEY is set, falls back to rule-based parser
// Auth: none  Rate: 30/min
export const POST = route(async ({ req }) => {
  const { q } = await readBody(req, Body);
  const ai = await parseSearchQuery(q);          // packages/ai/src/smartSearch.ts
  if (ai) return { filters, understood, engine: 'ai' };
  const rules = parseSmartQuery(q);              // lib/smartSearch.ts
  return { filters: rules.filters, understood: rules.understood, engine: 'rules' };
}, { auth: 'none', rate: { limit: 30, windowMs: 60_000, key: 'search' } });
